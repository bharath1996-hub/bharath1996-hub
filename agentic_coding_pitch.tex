\begin{frame}{What is Agentic Coding?}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{\textcolor{claudeOrange}{\faRobot\ Traditional Chatbox}}
\begin{itemize}
    \item Single prompt $\rightarrow$ single response
    \item Copy-paste code manually
    \item No file system access
    \item Cannot execute or test code
    \item You are the ``glue'' between steps
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{\textcolor{quantumBlue}{\faCogs\ Agentic Workflow (Claude Code)}}
\begin{itemize}
    \item Multi-step autonomous execution
    \item Reads, writes, edits files directly
    \item Runs terminal commands (git, pytest, etc.)
    \item Iterates on errors automatically
    \item \textbf{AI handles the entire workflow}
\end{itemize}
\end{column}
\end{columns}

\begin{frame}{The Workflow Difference: A Real Example}

\textbf{Task:} ``Implement a variational quantum eigensolver simulation''

\vspace{0.3cm}

\begin{columns}[T]
\begin{column}{0.45\textwidth}
\textcolor{claudeOrange}{\textbf{\faComments\ Chatbox Workflow}}
\begin{enumerate}
    \item Ask for VQE code snippet
    \item Copy code to file manually
    \item Run it $\rightarrow$ error!
    \item Go back, paste error
    \item Get fix, apply manually
    \item Repeat 5-10 times...
    \item Manually add tests
    \item Manually commit to git
\end{enumerate}
\textbf{Time:} Hours of context-switching
\end{column}

\begin{column}{0.45\textwidth}
\textcolor{quantumBlue}{\textbf{\faTerminal\ Agentic Workflow}}
\begin{enumerate}
    \item Give high-level task
    \item \textit{Agent explores codebase}
    \item \textit{Agent writes implementation}
    \item \textit{Agent runs, sees errors}
    \item \textit{Agent fixes autonomously}
    \item \textit{Agent writes tests \& runs them}
    \item \textit{Agent commits with message}
    \item You review the PR
\end{enumerate}
\textbf{Time:} Minutes of supervision
\end{column}
\end{columns}

\vspace{0.3cm}
\begin{center}
\colorbox{quantumBlue!20}{\parbox{0.8\textwidth}{\centering
\textbf{You shift from ``doing'' to ``directing'' --- focus on physics, not boilerplate.}
}}
\end{center}

\end{frame}
